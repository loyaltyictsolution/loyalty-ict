""" Loyalty ICT: removes the services tagline and the rotating service icons. Usage: python apply_edits.py index.html Makes a backup (index.html.bak) and rewrites index.html in place. """ import re, sys, shutil path = sys.argv[1] if len(sys.argv) > 1 else "index.html" src = open(path, encoding="utf-8").read() shutil.copyfile(path, path + ".bak") edits = [ ("Services tagline paragraph", r'[ \t]*

.*?

\n?', ""), ("services_sub default text", r"[ \t]*services_sub:'.*?',\n", ""), ("services_sub admin field", r"[ \t]*\['services_sub','Services intro paragraph','textarea'\],\n", ""), ("Rotating icons block", r'[ \t]*
\s*
.*?
\s*
\s*
\n?', ""), ("servInit cleanup", r"function servInit\(\)\{.*?\}\n", "function servInit(){ if(servCount()){ servGo(0); servResume(); } }\n"), ("startGalleryStream call", r"[ \t]*startGalleryStream\(\);[^\n]*\n", ""), ("Gallery editor in admin", r"[ \t]*\$\{galleryAdminHtml\(\)\}\n", ""), ] for name, pat, rep in edits: new, n = re.subn(pat, lambda m: rep, src, count=1, flags=re.S) print(("OK " if n else "NOT FOUND ") + name) src = new open(path, "w", encoding="utf-8").write(src) print("Done. Backup saved as", path + ".bak")
